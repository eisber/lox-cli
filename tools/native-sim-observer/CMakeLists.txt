cmake_minimum_required(VERSION 3.20)
project(native_sim_observer C ASM_MASM)
set(DynamoRIO_USE_LIBC OFF)
find_package(DynamoRIO 11.91 REQUIRED)
add_library(native_sim_observer SHARED observer.c)
configure_DynamoRIO_client(native_sim_observer)
use_DynamoRIO_extension(native_sim_observer drmgr)
target_compile_options(native_sim_observer PRIVATE /W4 /WX)
add_library(observer_noop SHARED isolation_client.c)
configure_DynamoRIO_client(observer_noop)
target_compile_options(observer_noop PRIVATE /W4 /WX)
add_library(observer_drmgr SHARED isolation_client.c)
configure_DynamoRIO_client(observer_drmgr)
use_DynamoRIO_extension(observer_drmgr drmgr)
target_compile_definitions(observer_drmgr PRIVATE ISOLATION_DRMGR)
target_compile_options(observer_drmgr PRIVATE /W4 /WX)
add_executable(observer_fixture fixture.c fixture_sentinels.asm)
target_compile_options(observer_fixture PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX;/Od;/Ob0>")
target_link_options(observer_fixture PRIVATE /INCREMENTAL:NO)
